<Steps>
<Step>

### Install

```bash
bun add @suku/core @suku/platform-bun zod
```

Any [Standard Schema](https://standardschema.dev) library works in place of Zod. Use
`@suku/platform-node` when you deploy on Node.

</Step>
<Step>

### A first route

```ts title="src/main.ts"
import { Suku, route } from '@suku/core';
import { bunAdapter } from '@suku/platform-bun';
import { z } from 'zod';

const hello = route.get('/hello/:name', {
  query: z.object({ shout: z.coerce.boolean().default(false) }),
  handle: ({ params, query }) => {
    const text = `Hello, ${params.name}`;
    return { message: query.shout ? text.toUpperCase() : text };
  },
});

const app = await Suku.create({ routes: [hello], adapter: bunAdapter() });
const server = await app.listen(3000);
console.log(`listening on ${server.url}`);
```

```bash
bun run src/main.ts
curl 'localhost:3000/hello/ada?shout=true'
# {"message":"HELLO, ADA"}
```

This style needs no decorators and no compiler options.

</Step>
<Step>

### Add a service and a routes class

Classes that receive services through their constructor need TypeScript's decorator metadata.
`UsersService` below takes nothing in its constructor, so it needs no decorator at all.

```json title="tsconfig.json"
{
  "compilerOptions": {
    "target": "ES2022",
    "module": "NodeNext",
    "moduleResolution": "NodeNext",
    "strict": true,
    "experimentalDecorators": true,
    "emitDecoratorMetadata": true
  }
}
```

```ts title="src/users.ts"
import { NotFoundException, Routes, feature, route, singleton } from '@suku/core';
import { z } from 'zod';

export class UsersService {
  private readonly users = new Map([[1, { id: 1, name: 'Ada Lovelace' }]]);

  find(id: number) {
    return this.users.get(id);
  }
}

@Routes('/users')
export class UsersRoutes {
  constructor(private readonly users: UsersService) {}

  findOne = route.get('/:id', {
    params: z.object({ id: z.coerce.number().int() }),
    handle: ({ params }) => {
      const user = this.users.find(params.id);
      if (!user) throw new NotFoundException(`User ${params.id} not found`);
      return user;
    },
  });
}

export const UsersFeature = feature('users', {
  routes: [UsersRoutes],
  services: [singleton(UsersService)],
});
```

```ts title="src/main.ts"
const app = await Suku.create({ features: [UsersFeature], routes: [hello], adapter: bunAdapter() });
```

<Callout type="error" title="Turn off type-only import rewriting">
  Lint rules that rewrite imports used only as types into `import type` erase the class that
  constructor injection needs. Disable Biome's `style/useImportType` and ESLint's
  `consistent-type-imports`. Suku reports the problem at startup as an
  `UnresolvableParameterError` naming the parameter.
</Callout>

</Step>
<Step>

### Run it

<Tabs items={['Bun', 'Node']}>
  <Tab value="Bun">
    Bun runs TypeScript and emits decorator metadata itself.

    ```bash
    bun run src/main.ts
    ```
  </Tab>
  <Tab value="Node">
    Swap `bunAdapter()` for `nodeAdapter()`. Node cannot execute decorators, so compile first
    with `tsc`, or bundle. Apps that use only plain routes can skip this.

    ```bash
    bun build src/main.ts --target node --outdir dist
    node dist/main.js
    ```
  </Tab>
</Tabs>

</Step>
