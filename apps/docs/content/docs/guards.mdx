---
title: Guards
description: Allow or reject a request before the handler runs.
icon: ShieldCheck
---

A guard returns `true` to let the request through. `false` becomes
`403 Forbidden resource`. Throw an `HttpException` for any other status.

```ts
import type { Guard } from '@suku/core';

@Injectable({ lifetime: 'scoped' })
export class AuthGuard implements Guard {
  constructor(
    private readonly ctx: RequestContext,
    private readonly tokens: TokenService,
  ) {}

  async canActivate() {
    const header = this.ctx.headers.get('authorization');
    if (!header) throw new UnauthorizedException();
    this.ctx.set('user', await this.tokens.verify(header));
    return true;
  }
}
```

A function works for simple checks.

```ts
@UseGuards((ctx) => ctx.headers.get('x-api-key') === process.env.API_KEY)
```

## Roles with metadata

`@SetMetadata` attaches a value to a class or handler. Guards read it with
`ctx.getMetadata()`. A value on the method overrides the one on the class.

```ts
export const Roles = (...roles: string[]) => SetMetadata('roles', roles);

@Injectable({ lifetime: 'scoped' })
export class RolesGuard implements Guard {
  constructor(private readonly ctx: RequestContext) {}

  canActivate() {
    const required = this.ctx.getMetadata<string[]>('roles') ?? [];
    const user = this.ctx.get<{ role: string }>('user');
    return required.length === 0 || (user !== undefined && required.includes(user.role));
  }
}

@Controller('reports')
@UseGuards(AuthGuard, RolesGuard)
@Roles('user')
export class ReportsController {
  @Get()
  list() {}

  @Delete(':id')
  @Roles('admin')
  remove() {}
}
```

Guards run in order: global, class, method. The first to refuse ends the request.
