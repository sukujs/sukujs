```ts title="main.ts"
import { Routes, Suku, feature, route, singleton } from '@sukujs/core';
import { bunAdapter } from '@sukujs/platform-bun';
import { z } from 'zod';

class UsersService {
  find(id: number) {
    return { id, name: 'Ada' };
  }
}

@Routes('/users')
class UsersRoutes {
  constructor(private readonly users: UsersService) {}

  findOne = route.get('/:id', {
    params: z.object({ id: z.coerce.number() }),
    // params.id is a number
    handle: ({ params }) => this.users.find(params.id),
  });
}

const UsersFeature = feature('users', {
  routes: [UsersRoutes],
  services: [singleton(UsersService)],
});

const app = await Suku.create({
  features: [UsersFeature],
  adapter: bunAdapter(),
});
await app.listen(3000);
```
