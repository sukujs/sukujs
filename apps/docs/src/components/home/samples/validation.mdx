```ts title="users.routes.ts"
import { z } from 'zod';

const CreateUser = z.object({
  name: z.string().min(2),
  email: z.email(),
});

route.post('/users', {
  body: CreateUser,
  services: { users: UsersService },
  // body is { name: string; email: string }
  handle: ({ body, users }) => users.create(body),
});
```
